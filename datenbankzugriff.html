// Erster Schritt: Verbindung zur MySQL-Datenbank [dbpersons] herstellen und dann
// trennen. Portiert das Python-Skript [mysql_01] des Python/Flask-Kurses.
import mysql from 'mysql2/promise';

// Verbindung zur MySQL-Datenbank [dbpersons]
// die Identität des Benutzers ist (admpersons, nobody)
const USER = 'admpersons';
const PWD = 'nobody';
const HOST = 'localhost';
const DATABASE = 'dbpersons';

// los geht's
let connection: mysql.Connection | undefined;
try {
  console.log('Connecting to the MySQL DBMS...');
  // Verbindung
  connection = await mysql.createConnection({ host: HOST, user: USER, password: PWD, database: DATABASE });
  // Fortsetzung
  console.log(
    `Successful MySQL connection to database=${DATABASE}, host=${HOST} as user=${USER}, passwd=${PWD}`,
  );
} catch (error) {
  // den Fehler anzeigen
  console.log(`The following error occurred: ${error}`);
} finally {
  // die Verbindung schließen, falls sie geöffnet wurde
  if (connection) {
    await connection.end();
  }
}

    Kommentieren wir diesen Code:

    • Zeile 16: [mysql.createConnection({ host: HOST, user: USER, password: PWD, database: DATABASE })] – öffnet eine einzelne, asynchrone Verbindung – createConnection gibt ein Promise<Connection> zurück, daher das await; dies ist das TypeScript-Äquivalent zu mysql.connector.connect(...) auf der Python-Seite;
    • Zeile 29: [connection.end()] – schließt die Verbindung im finally-Block ordnungsgemäß, unabhängig davon, ob ein Fehler aufgetreten ist – die Verbindung wird außerhalb des try -Blocks deklariert (Typ mysql.Connection | undefined), damit sie auf dieser Ebene zugänglich bleibt.

    // databases/mysql/mysql-02.ts
// In diesem neuen Skript ist die Datenbankverbindung in einer Funktion isoliert.
// Portiert das Python-Skript [mysql_02].
import mysql from 'mysql2/promise';

// verbindet sich mit der Datenbank [database] auf dem Server [host] (login, pwd) und trennt sich dann wieder
// wirft eine Exception, wenn es ein Problem gibt
async function connect(host: string, database: string, login: string, pwd: string): Promise<void> {
  let cnx: mysql.Connection | undefined;
  try {
    // Verbindung
    cnx = await mysql.createConnection({ host, user: login, password: pwd, database });
    console.log(
      `Successful connection to database=${database}, host=${host} as user=${login}, passwd=${pwd}`,
    );
  } finally {
    // die Verbindung schließen, falls sie geöffnet wurde
    if (cnx) {
      await cnx.end();
      console.log('Successful disconnection\n');
    }
  }
}

// main
// Verbindungsdaten
const USER = 'admpersons';
const PASSWD = 'nobody';
const HOST = 'localhost';
const DATABASE = 'dbpersons';

// Verbindung eines vorhandenen Benutzers
try {
  await connect(HOST, DATABASE, USER, PASSWD);
} catch (error) {
  // den Fehler anzeigen
  console.log(String(error));
}

// Verbindung eines nicht vorhandenen Benutzers
try {
  await connect(HOST, DATABASE, 'xx', 'xx');
} catch (error) {
  // den Fehler anzeigen
  console.log(String(error));
}

    Kommentieren wir diesen Code:

    • Zeile 9: [async function connect(host, database, login, pwd): Promise<void>] – eine async-Funktion, die nichts zurückgibt, aber eine Ausnahme auslösen kann – sie fängt Verbindungsfehler nicht selbst ab, sondern lässt sie an den Aufrufer weiterlaufen;
    • Zeile 40: [await connect(HOST, DATABASE, 'xx', 'xx')] – der zweite Aufruf mit einem Benutzer „xx“ , der auf der MySQL-Seite nicht existiert, löst eine Ausnahme aus, die der umschließende catch-Block anzeigt.

    Versionsspezifisches Problem: Bei einem aktuellen MySQL-Server (ab Version 8.4) kann dieser zweite Aufruf eine andere Meldung erzeugen: „Error: Plugin 'mysql_native_password' is not loaded“ statt „Access denied for user 'xx'@'localhost'.“ Die Ursache hat nichts mit dem Skript zu tun: Um nicht preiszugeben, ob ein Konto existiert oder nicht (Schutz vor Enumeration), simuliert MySQL bei der Anmeldung mit einem unbekannten Benutzer einen Authentifizierungsaustausch – und diese Simulation ist fest auf das veraltete Plugin mysql_native_password verdrahtet, das seit MySQL 8.4 standardmäßig deaktiviert ist. Das von diesem Skript demonstrierte Prinzip (eine bei ungültigen Zugangsdaten ausgelöste und abgefangene Ausnahme) bleibt davon unberührt; um die genau angezeigte Meldung wiederherzustellen, fügen Sie mysql\_native\_password=ON unterhalb von [mysqld] in der Datei my.ini des Servers hinzu und starten den Server anschließend neu.

// Da wir nun wissen, wie man sich verbindet, senden wir SQL-Anweisungen über die Verbindung:
// wir erstellen die Tabelle [persons] der Datenbank [dbpersons] (neu).
// Portiert das Python-Skript [mysql_03].
import mysql from 'mysql2/promise';

// führt eine Update-Abfrage über die Verbindung aus
async function executeSql(connection: mysql.Connection, query: string): Promise<void> {
  await connection.query(query);
}

// main
// Verbindungsdaten
const ID = 'admpersons';
const PWD = 'nobody';
const HOST = 'localhost';
const DATABASE = 'dbpersons';

// wir gehen Schritt für Schritt vor
let connection: mysql.Connection;
try {
  // Verbindung - mysql2 führt jede Abfrage in einer eigenen Transaktion aus,
  // die standardmäßig automatisch committet wird (Entsprechung von AUTOCOMMIT=True auf der Python-Seite)
  connection = await mysql.createConnection({ host: HOST, user: ID, password: PWD, database: DATABASE });
} catch (error) {
  console.log(`The following error occurred: ${error}`);
  process.exit(1);
}

// Löschen der Tabelle persons, falls sie existiert
// falls sie nicht existiert, tritt ein Fehler auf - wir ignorieren ihn
try {
  await executeSql(connection, 'drop table persons');
} catch {
  // ignoriert
}

// Erstellen der Tabelle persons
const query =
  'create table persons (id int PRIMARY KEY, first_name varchar(30) NOT NULL, last_name varchar(30) NOT NULL, ' +
  'age integer NOT NULL, unique(last_name,first_name))';
try {
  await executeSql(connection, query);
  console.log(`${query}: query succeeded`);
} catch (error) {
  console.log(`The following error occurred: ${error}`);
} finally {
  await connection.end();
}

    • Zeile 9: [await connection.query(query)] — connection.query() führt einen SQL-String direkt ohne Parameter aus – die sicherere, parametrisierte Variante mit execute() sehen wir im Skript [mysql-05];
    • Zeile 28: [await executeSql(connection, 'drop table persons')] – es wird versucht, die Tabelle innerhalb eines leeren try/catch-Blocks zu löschen: Existiert die Tabelle noch nicht, wird der MySQL-Fehler einfach ignoriert.

    // databases/mysql/mysql-module.ts
// Wiederverwendbare Funktionen zum Ausführen einer Liste (oder einer Datei) von SQL-Befehlen
// über eine offene MySQL-Verbindung. Portiert das Python-Modul [mysql_module].
import type { Connection, FieldPacket, ResultSetHeader, RowDataPacket } from 'mysql2/promise';
import { readFile } from 'node:fs/promises';

// zeigt das Ergebnis eines SQL-Befehls an (Ergebnis eines mysql2-query())
// - gibt es ein Array [fields], war es ein SELECT: wir zeigen die Spalten
//   und dann die Zeilen an;
// - andernfalls war es eine Änderungsanweisung (INSERT/UPDATE/DELETE/DDL): wir
//   zeigen die Anzahl der geänderten Zeilen an (Entsprechung von cursor.rowcount).
function displayInfo(rows: RowDataPacket[] | ResultSetHeader, fields: FieldPacket[] | undefined): void {
  if (fields && fields.length > 0) {
    // es war ein SELECT — fields beschreibt die angeforderten Spalten
    const titles = fields.map((f) => f.name);
    console.log(titles.join(', '));
    console.log('*'.repeat(titles.join(', ').length));
    for (const row of rows as RowDataPacket[]) {
      console.log(titles.map((t) => row[t]));
    }
    console.log('*'.repeat(titles.join(', ').length));
  } else {
    // keine Spalten beschrieben — Änderungsanweisung
    const header = rows as ResultSetHeader;
    console.log(`number of modified rows: ${header.affectedRows}`);
  }
}

// verwendet die offene Verbindung [connection]
// führt über diese Verbindung die SQL-Befehle aus der Liste [sqlCommands] aus
// (ein Befehl pro Element; leere Zeilen oder Zeilen, die mit # beginnen, werden ignoriert)
// - bei verbose=true folgt auf jede Ausführung einer SQL-Anweisung eine Anzeige,
//   die ihren Erfolg oder Misserfolg angibt;
// - bei stop=true stoppt die Funktion beim 1. aufgetretenen Fehler, andernfalls
//   führt sie alle Befehle aus;
// - bei withTransaction=true macht jeder Fehler alle zuvor ausgeführten SQL-Anweisungen
//   rückgängig (Rollback); andernfalls wird jeder Befehl
//   unabhängig committet (Standardverhalten von mysql2);
// die Funktion gibt ein Array [error1, error2, ...] zurück
export async function executeListOfCommands(
  connection: Connection,
  sqlCommands: string[],
  verbose = false,
  stop = true,
  withTransaction = true,
): Promise<string[]> {
  const errors: string[] = [];
  if (withTransaction) {
    await connection.beginTransaction();
  }
  try {
    for (const rawCommand of sqlCommands) {
      // Leerzeichen am Anfang und Ende des aktuellen Befehls entfernen
      const command = rawCommand.trim();
      // leerer Befehl oder Kommentar? weiter zum nächsten
      if (command === '' || command[0] === '#') {
        continue;
      }
      // Ausführung des aktuellen Befehls
      try {
        const [rows, fields] = await connection.query(command);
        // kein Fehler
        if (verbose) {
          console.log(`[${command}]: Successful execution`);
        }
        displayInfo(rows as RowDataPacket[], fields);
      } catch (error) {
        const msg = `${command}: Error (${error})`;
        errors.push(msg);
        if (verbose) {
          console.log(msg);
        }
        // halten wir an?
        if (withTransaction || stop) {
          return errors;
        }
      }
    }
    return errors;
  } finally {
    // die Transaktion committen / zurückrollen, falls es eine gibt
    if (withTransaction) {
      if (errors.length > 0) {
        await connection.rollback();
      } else {
        await connection.commit();
      }
    }
  }
}

// verwendet die offene Verbindung [connection]
// führt über diese Verbindung die SQL-Befehle aus der
// Textdatei [sqlFilename] aus (ein Befehl pro Zeile)
export async function executeFileOfCommands(
  connection: Connection,
  sqlFilename: string,
  verbose = false,
  stop = true,
  withTransaction = true,
): Promise<string[]> {
  try {
    const content = await readFile(sqlFilename, 'utf-8');
    return await executeListOfCommands(connection, content.split('\n'), verbose, stop, withTransaction);
  } catch (error) {
    return [`The file ${sqlFilename} could not be processed: ${error}`];
  }
}

    Kommentieren wir diesen Code:

    • Zeile 45: [if (withTransaction) { await connection.beginTransaction(); }] – mysql2 befindet sich standardmäßig im AUTOCOMMIT-Modus (jede Abfrage wird einzeln bestätigt) – beginTransaction() schaltet die Verbindung explizit in den Transaktionsmodus, genau wie es connexion.start_transaction() auf der Python-Seite täte;
    • Zeile 76: [if (withTransaction || stop) { return errors; }] – im Transaktionsmodus beendet der erste Fehler die Schleife sofort – weiterzumachen wäre sinnlos, da die gesamte Transaktion ohnehin zurückgerollt wird;
    • Zeile 82: [if (errors.length > 0) { await connection.rollback(); } else { await connection.commit(); }] – globaler Commit oder Rollback, je nachdem, ob ein Fehler aufgetreten ist oder nicht – siehe jedoch die Falle, auf die im folgenden Skript [mysql-04] hingewiesen wird.

// Parametrisierte Abfragen: einzelnes vorbereitetes Einfügen, dann Masseneinfügung.
// Portiert das Python-Skript [mysql_05]. Hier erstellen wir die Tabelle [persons] neu mit
// dem Schema mit einer Spalte [id] (wie im Skript mysql-03), damit dieses Skript
// unabhängig von dem Zustand ist, den mysql-04 hinterlässt.
import mysql from 'mysql2/promise';

// die Identität des Benutzers
const ID = 'admpersons';
const PWD = 'nobody';
// der Rechner des DBMS
const HOST = 'localhost';
// Identität der Datenbank
const DATABASE = 'dbpersons';

type Person = [id: number, lastName: string, firstName: string, age: number];

// Liste von Personen (id, lastName, firstName, age)
const persons: Person[] = [];
for (let i = 0; i < 5; i++) {
  persons.push([i, `l0${i}`, `f0${i}`, i + 10]);
}
persons.push([40, "d'Aboot", "Y'éna", 18]);

// eine weitere Liste von Personen
const otherPersons: Person[] = [];
for (let i = 0; i < 5; i++) {
  otherPersons.push([i + 100, `l1${i}`, `f1${i}`, i + 20]);
}
otherPersons.push([200, "d'Aboot", "F'ilhem", 34]);

// Zugriff auf das DBMS
let connection: mysql.Connection | undefined;
try {
  // Verbindung
  connection = await mysql.createConnection({ host: HOST, user: ID, password: PWD, database: DATABASE });

  // (Neu-)Erstellung der Tabelle mit dem Schema mit [id]-Spalte (siehe mysql-03.ts)
  await connection.query('drop table if exists persons');
  await connection.query(
    'create table persons (id int PRIMARY KEY, first_name varchar(30) NOT NULL, last_name varchar(30) NOT NULL, ' +
      'age integer NOT NULL, unique(last_name,first_name))',
  );

  // Löschen der vorhandenen Datensätze
  await connection.query('delete from persons');
  // Einfügen Person für Person mit einem vorbereiteten Statement
  for (const person of persons) {
    const [id, lastName, firstName, age] = person;
    await connection.execute('insert into persons(id,last_name,first_name,age) values(?,?,?,?)', [id, lastName, firstName, age]);
  }
  // Masseneinfügung einer Liste von Personen — Entsprechung von executemany:
  // mysql2 bietet connection.query('insert ... values ?', [[...],[...],...])
  // das eine einzige mehrzeilige Abfrage erzeugt
  const values = otherPersons.map(([id, lastName, firstName, age]) => [id, lastName, firstName, age]);
  await connection.query('insert into persons(id,last_name,first_name,age) values ?', [values]);

  // Committen der Transaktion — hier wurde jede Abfrage bereits automatisch committet
  // (mysql2 befindet sich standardmäßig im AUTOCOMMIT-Modus, wie der im
  // Skript mysql-03 erwähnte Standardmodus; es gibt also nichts explizit zu committen)
  console.log(`${persons.length + otherPersons.length} persons inserted.`);
} catch (error) {
  console.log(`The following error occurred: ${error}`);
} finally {
  if (connection) {
    await connection.end();
  }
}

    7.2. Verwendung des ORM TypeORM

    Wir behandeln nun ORM, angelehnt an das Kapitel „Verwendung des ORM SQLAlchemy“ (Kapitel 19) des Python/Flask-Kurses. Das Prinzip hinter ORM (Object-Relational Mapping) besteht darin, Objekte der Sprache zu manipulieren, statt SQL zu schreiben: Eine Klasse entspricht einer Tabelle, eine Instanz einer Zeile, ein Attribut einer Spalte und eine Beziehung zwischen Klassen (Assoziation, Komposition) einem Fremdschlüssel.

Von beiden ähnelt TypeORM im Stil am stärksten SQLAlchemy (dekorierte Klassen, per Methoden abgefragte Repositorys); Prisma hingegen erzeugt aus einer separaten, deklarativen Schemadatei einen vollständig typisierten Client.

// berechnet die Statistiken (gewichteter Durchschnitt, Minimum, Maximum) eines Schülers
async function studentStats(studentId: number): Promise<void> {
  const studentRepo = AppDataSource.getRepository(Student);
  // wir laden den Schüler mit seinen Noten und für jede Note das zugehörige
  // Fach (um den Koeffizienten zu erhalten) — eine einzige Abfrage mit zwei
  // Joins, dank der verschachtelten Option `relations`
  const student = await studentRepo.findOneOrFail({
    where: { id: studentId },
    relations: { grades: { subject: true } },
  });

  console.log(`Statistics of student {"id":${student.id},"lastName":"${student.lastName}","firstName":"${student.firstName}"}`);

  if (student.grades.length === 0) {
    console.log('  no grades.');
    return;
  }

  let weightedSum = 0;
  let coefficientsSum = 0;
  let min = Number.POSITIVE_INFINITY;
  let max = Number.NEGATIVE_INFINITY;

  for (const grade of student.grades) {
    // explizite Umwandlung: `grade.value` und `grade.subject.coefficient`
    // sind zur Laufzeit Zeichenketten, trotz ihres deklarierten Typs `number`
    const value = Number(grade.value);
    const coefficient = Number(grade.subject.coefficient);
    weightedSum += value * coefficient;
    coefficientsSum += coefficient;
    min = Math.min(min, value);
    max = Math.max(max, value);
    console.log(`  grade=${value.toFixed(2)} subject="${grade.subject.name}" (coefficient ${coefficient.toFixed(2)})`);
  }

  const average = weightedSum / coefficientsSum;
  console.log(`  weighted average=${average.toFixed(2)}, min=${min.toFixed(2)}, max=${max.toFixed(2)}`);
}

await AppDataSource.initialize();
try {
  // Schüler id=11: Noten subject1=10 (Koeff. 1.0), subject2=6 (Koeff. 2.0)
  // gewichteter Durchschnitt = (10*1.0 + 6*2.0) / (1.0+2.0) = 22/3 = 7.33
  await studentStats(11);
  console.log();
  // Schüler id=21: Noten subject1=12 (Koeff. 1.0), subject2=8 (Koeff. 2.0)
  await studentStats(21);
  console.log();
  // Schüler id=32: Noten subject1=14 (Koeff. 1.0), subject2=10 (Koeff. 2.0)
  await studentStats(32);
  console.log();
  // Schüler id=42: Noten subject1=16 (Koeff. 1.0), subject2=12 (Koeff. 2.0)
  await studentStats(42);
} finally {
  await AppDataSource.destroy();
}

    TypeORM ermöglicht es, Lese-/Schreiboperationen ohne SQL zu schreiben, mit dekorierten Entitäten, die SQLAlchemy im Geiste ähneln – um den Preis eines wesentlichen Designunterschieds, den man kennen sollte (kein implizites Lazy Loading von Beziehungen: alle nicht ausdrücklich angeforderten Beziehungen werden weggelassen) und einer Werkzeugfalle, die speziell bei der Ausführung mit TSX auftritt (Spaltentypen müssen explizit angegeben werden).     7.3. Verwendung des ORM Prisma

    Zweites ORM, mit demselben Beispiel wie im vorherigen Kapitel (die Datenbanken dbpersons und dbschool), um die beiden Ansätze zu vergleichen. Prisma unterscheidet sich von TypeORM in seinem Entwurf: Statt Klassen zu annotieren, wird das Datenbankschema in einer separaten Datei im eigenen deklarativen Format von Prisma beschrieben, aus der der Befehl prisma generate einen vollständig typisierten Client erzeugt.

    Wichtig: Zum Zeitpunkt der Erstellung dieses Kapitels liegt Prisma ORM in Version 7 vor, die gegenüber den früheren Versionen (5 und 6) wesentliche interne Änderungen erfahren hat: Die Query Engine, früher ein separat heruntergeladenes, in Rust kompiliertes Binary, ist nun ein WebAssembly-Modul, das direkt im npm-Paket eingebettet ist, kombiniert mit einem Driver-Adapter (hier @prisma/adapter-mariadb), der Abfragen an einen herkömmlichen JavaScript-Treiber weiterleitet. Das allgemeine Prinzip bleibt jedoch dasselbe wie bei früheren Prisma-Versionen, und das Folgende gilt daher auch für diese.

    Hinweis zu dieser Umgebung: Der Befehl npx prisma generate , der den TypeScript-Client aus der Datei schema.prisma erzeugt, benötigt eine Prisma-spezifische Netzwerkressource (binaries.prisma.sh), um eine interne Komponente seiner CLI abzurufen – eine Ressource, auf die die Netzwerkrichtlinie der Laufzeitumgebung, in der dieser Kurs vorbereitet wurde, keinen Zugriff erlaubte. Die Skripte dieses Kapitels wurden daher mit derselben Sorgfalt wie die vorherigen geschrieben und geprüft, konnten in dieser Umgebung jedoch nicht ausgeführt oder verifiziert werden – anders als jene der beiden vorherigen Kapitel, die alle ausgeführt und deren Ergebnisse überprüft wurden. Auf einem Rechner mit normalem Internetzugang (dem typischen Fall) läuft npx prisma generate ohne besondere Probleme, und die Skripte sollten die unten gezeigten „erwarteten“ Ausgaben liefern, die aus denselben, im vorherigen Kapitel bereits verifizierten Daten berechnet wurden.

// Abfragen mit Joins (`include`) auf [dbschool]. Verwendet genau dasselbe
// Beispiel wie databases/typeorm/typeorm-03-joined-queries.ts — vergleichen Sie die
// beiden Dateien: Die Logik ist identisch, nur die API ändert sich (`include`
// statt `relations`, `findUniqueOrThrow`/`findFirstOrThrow` statt
// `findOneOrFail`).
import { PrismaMariaDb } from '@prisma/adapter-mariadb';
import { PrismaClient } from '../../generated/prisma-school/client.js';

const adapter = new PrismaMariaDb({
  host: 'localhost',
  user: 'admschool',
  password: 'pwdschool',
  database: 'dbschool',
});
const prisma = new PrismaClient({ adapter });

try {
  // Schüler nach ID, mit seiner Klasse (expliziter Join über `include`)
  console.log('student id=11 -----------');
  const student11 = await prisma.student.findUniqueOrThrow({ where: { id: 11 }, include: { schoolClass: true } });
  console.log(`student={"id":${student11.id},"lastName":"${student11.lastName}","firstName":"${student11.firstName}"}`);
  console.log(`student's class: {"id":${student11.schoolClass.id},"name":"${student11.schoolClass.name}"}`);

  // die Schüler derselben Klasse (neue Abfrage, mit eigenem Join)
  console.log('students in the same class:');
  const studentsClass = await prisma.schoolClass.findUniqueOrThrow({
    where: { id: student11.schoolClass.id },
    include: { students: true },
  });
  for (const s of studentsClass.students) {
    console.log(`student={"id":${s.id},"lastName":"${s.lastName}","firstName":"${s.firstName}"}`);
  }

  // ein Schüler nach seinem Nachnamen
  console.log("student lastName='lastname2' -----------");
  const byName = await prisma.student.findFirstOrThrow({ where: { lastName: 'lastname2' }, include: { schoolClass: true } });
  console.log(`student={"id":${byName.id},"lastName":"${byName.lastName}","firstName":"${byName.firstName}"}`);
  console.log(`student's class: {"id":${byName.schoolClass.id},"name":"${byName.schoolClass.name}"}`);

  // die Noten eines Schülers, mit dem Fach jeder Note — verschachteltes `include`,
  // Äquivalent zu `relations: { grades: { subject: true } }` von TypeORM
  console.log('grades of student id=11 -----------');
  const studentWithGrades = await prisma.student.findUniqueOrThrow({
    where: { id: 11 },
    include: { grades: { include: { subject: true } } },
  });
  for (const grade of studentWithGrades.grades) {
    console.log(`grade={"id":${grade.id},"value":${grade.value}}, subject={"name":"${grade.subject.name}"}`);
  }

  // die Schüler einer Klasse, nach Klassenname
  for (const className of ['class1', 'class2']) {
    console.log(`students of class name='${className}' -----------`);
    const schoolClass = await prisma.schoolClass.findUniqueOrThrow({ where: { name: className }, include: { students: true } });
    for (const s of schoolClass.students) {
      console.log(`{"id":${s.id},"lastName":"${s.lastName}","firstName":"${s.firstName}"}`);
    }
  }

  // die Noten in einem Fach, nach Fachname
  for (const subjectName of ['subject1', 'subject2']) {
    console.log(`subject with name='${subjectName}' -----------`);
    const subject = await prisma.subject.findUniqueOrThrow({ where: { name: subjectName }, include: { grades: true } });
    console.log(`subject={"name":"${subject.name}","coefficient":${subject.coefficient}}`);
    console.log('Grades in the subject: ');
    for (const grade of subject.grades) {
      console.log(`{"id":${grade.id},"value":${grade.value}}`);
    }
  }
} finally {
  await prisma.$disconnect();
}

    // databases/prisma/prisma-04-stats.ts
// Berechnet für einen gegebenen Schüler seinen mit den Koeffizienten der
// Fächer gewichteten Durchschnitt sowie seine minimale und maximale Note. Verwendet dasselbe Beispiel wie
// databases/typeorm/typeorm-04-stats.ts.
//
// Fallstrick, der sich von dem bei TypeORM unterscheidet: Prisma liefert die
// `Decimal`-Spalten (hier `grade.value` und `subject.coefficient`) nicht als
// Zeichenketten zurück, sondern als Instanzen der Klasse `Prisma.Decimal`
// (bereitgestellt von der Bibliothek decimal.js), um keine
// Dezimalgenauigkeit zu verlieren. `typeof grade.value` ist `'object'`, weder `'number'`
// noch `'string'`. Eine Berechnung wie `grade.value * subject.coefficient`
// würde fast zufällig funktionieren (`Decimal` überschreibt `valueOf`), aber
// die bewährte Vorgehensweise — und die unten verwendete — ist, vor jeder
// Gleitkommaberechnung explizit `.toNumber()` aufzurufen.
import { PrismaMariaDb } from '@prisma/adapter-mariadb';
import { PrismaClient } from '../../generated/prisma-school/client.js';

const adapter = new PrismaMariaDb({
  host: 'localhost',
  user: 'admschool',
  password: 'pwdschool',
  database: 'dbschool',
});
const prisma = new PrismaClient({ adapter });

// berechnet die Statistiken (gewichteter Durchschnitt, Minimum, Maximum) eines Schülers
async function studentStats(studentId: number): Promise<void> {
  // eine einzige Abfrage mit zwei Join-Ebenen (Noten, dann das Fach
  // jeder Note), um den Koeffizienten jedes Fachs zu erhalten
  const student = await prisma.student.findUniqueOrThrow({
    where: { id: studentId },
    include: { grades: { include: { subject: true } } },
  });

  console.log(`Statistics of student {"id":${student.id},"lastName":"${student.lastName}","firstName":"${student.firstName}"}`);

  if (student.grades.length === 0) {
    console.log('  no grades.');
    return;
  }

  let weightedSum = 0;
  let coefficientsSum = 0;
  let min = Number.POSITIVE_INFINITY;
  let max = Number.NEGATIVE_INFINITY;

  for (const grade of student.grades) {
    // explizite Umwandlung: `grade.value` und `grade.subject.coefficient`
    // sind `Prisma.Decimal`-Werte, keine `number`-Werte
    const value = grade.value.toNumber();
    const coefficient = grade.subject.coefficient.toNumber();
    weightedSum += value * coefficient;
    coefficientsSum += coefficient;
    min = Math.min(min, value);
    max = Math.max(max, value);
    console.log(`  grade=${value.toFixed(2)} subject="${grade.subject.name}" (coefficient ${coefficient.toFixed(2)})`);
  }

  const average = weightedSum / coefficientsSum;
  console.log(`  weighted average=${average.toFixed(2)}, min=${min.toFixed(2)}, max=${max.toFixed(2)}`);
}

try {
  // Schüler id=11: Noten subject1=10 (Koeff. 1.0), subject2=6 (Koeff. 2.0)
  // gewichteter Durchschnitt = (10*1.0 + 6*2.0) / (1.0+2.0) = 22/3 = 7.33
  await studentStats(11);
  console.log();
  await studentStats(21);
  console.log();
  await studentStats(32);
  console.log();
  await studentStats(42);
} finally {
  await prisma.$disconnect();
}

    7.3.6. Fazit

    Bei demselben Beispiel erzeugen TypeORM und Prisma Code von vergleichbarer Länge und Lesbarkeit – die Unterschiede sind hauptsächlich stilistischer Art: dekorierte Klassen und ein Repository bei TypeORM (näher an SQLAlchemy und Spring Data JPA) sowie ein separates deklaratives Schema und ein vollständig generierter Client bei Prisma (im Geiste näher an einem Werkzeug wie Alembic in Kombination mit einem Codegenerator). Beide vermeiden es, SQL zu schreiben, und beide teilen dieselbe Designentscheidung – kein implizites Lazy Loading von Beziehungen –, die sie von SQLAlchemy unterscheidet. Die Wahl zwischen beiden hängt in einem realen Projekt vor allem vom Ökosystem (Prisma Studio, Prisma-Migrationen usw.) und von den Präferenzen des Teams ab.
