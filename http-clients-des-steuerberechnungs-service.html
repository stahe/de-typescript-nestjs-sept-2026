// Importe
import { AxiosInstance, AxiosRequestConfig, AxiosResponse } from 'axios';
import qs from 'qs'

class Dao1 {

  private axios: AxiosInstance;
  private sessionCookieName: string;
  private sessionCookie: string;

  // Konstruktor
  constructor(axios: AxiosInstance) {
    // Bibliothek axios, die für die HTTP-Anfragen verwendet wird
    this.axios = axios;
    // Session-Cookie
    this.sessionCookieName = "PHPSESSID";
    this.sessionCookie = '';
  }

  // Sitzung initialisieren
  async initSession(): Promise<any> {
    // Optionen der HTTP-Anfrage [get /main.php?action=init-session&type=json]
    const options: AxiosRequestConfig = {
      // [TypeScript-Aktualisierung] URL in [options.url] enthalten (siehe getRemoteData)
      url: 'main.php',
      method: "GET",
      // URL-Parameter
      params: {
        action: 'init-session',
        type: 'json'
      }
    };
    // Ausführung der HTTP-Anfrage
    return await this.getRemoteData(options);
  }

  async authenticateUser(user: string, password: string): Promise<any> {
    // Optionen der HTTP-Anfrage [post /main.php?action=authenticate-user]
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "POST",
      headers: {
        'Content-type': 'application/x-www-form-urlencoded',
      },
      // Body des POST
      data: qs.stringify({
        user: user,
        password: password
      }),
      // URL-Parameter
      params: {
        action: 'authenticate-user'
      }
    };
    // Ausführung der HTTP-Anfrage
    return await this.getRemoteData(options);
  }

  // Steuerberechnung
  async calculateTax(married: string, children: number, salary: number): Promise<any> {
    // Optionen der HTTP-Anfrage
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "POST",
      headers: {
        'Content-type': 'application/x-www-form-urlencoded',
      },
      // Body des POST
      data: qs.stringify({
        married: married,
        children: children,
        salary: salary
      }),
      // URL-Parameter
      params: {
        action: 'calculate-tax'
      }
    };
    // Ausführung der HTTP-Anfrage
    const data = await this.getRemoteData(options);
    // Ergebnis
    return data;
  }

  // Liste der Simulationen
  async listSimulations(): Promise<any> {
    // Optionen der HTTP-Anfrage
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "GET",
      // URL-Parameter
      params: {
        action: 'list-simulations'
      },
    };
    // Ausführung der HTTP-Anfrage
    const data = await this.getRemoteData(options);
    // Ergebnis
    return data;
  }

  // eine Simulation löschen
  async deleteSimulation(index: number): Promise<any> {
    // Optionen der HTTP-Anfrage
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "GET",
      // URL-Parameter
      params: {
        action: 'delete-simulation',
        number: index
      },
    };
    // Ausführung der HTTP-Anfrage
    const data = await this.getRemoteData(options);
    // Ergebnis
    return data;
  }

  async getRemoteData(options: AxiosRequestConfig): Promise<any> {
    // für das Session-Cookie
    if (!options.headers) {
      options.headers = {} as any;
    }
    (options.headers as any).Cookie = this.sessionCookie;
    // Ausführung der HTTP-Anfrage
    let response: AxiosResponse;
    try {
      // asynchrone Anfrage
      response = await this.axios.request(options);
    } catch (error: any) {
      // der Parameter [error] ist eine Exception-Instanz - sie kann verschiedene Formen annehmen
      if (error.response) {
        // die Antwort des Servers steht in [error.response]
        response = error.response;
      } else {
        // wir werfen den Fehler erneut
        throw error;
      }
    }
    // response ist die gesamte HTTP-Antwort des Servers (HTTP-Header + die eigentliche Antwort)
    // wir rufen das Session-Cookie ab, falls vorhanden
    const setCookie = response.headers['set-cookie'];
    if (setCookie) {
      // setCookie ist ein Array
      // wir suchen das Session-Cookie in diesem Array
      let found = false;
      let i = 0;
      while (!found && i < setCookie.length) {
        // wir suchen das Session-Cookie
        const results = RegExp('^(' + this.sessionCookieName + '.+?);').exec(setCookie[i]);
        if (results) {
          // wir speichern das Session-Cookie
          // eslint-disable-next-line require-atomic-updates
          this.sessionCookie = results[1];
          // wir haben es gefunden
          found = true;
        } else {
          // nächstes Element
          i++;
        }
      }
    }
    // die Antwort des Servers steht in [response.data]
    return response.data;
  }
}

// Export der Klasse
export default Dao1;

// asynchrone Funktion [main]
async function main(): Promise<void> {
  // Konfiguration von axios
  axios.defaults.timeout = 2000;
  axios.defaults.baseURL = 'http://localhost:3000';
  // Instanziierung der Schicht [dao]
  const dao = new Dao(axios);
  // Verwendung der Schicht [dao]
  try {
    // Sitzung initialisieren

    let response = await dao.initSession();
    log(response);
    // Authentifizierung
    log("-----------authenticate-user");
    response = await dao.authenticateUser("admin", "admin");
    log(response);
    // Steuerberechnungen
    log("-----------calculate-tax x 3");
    response = await Promise.all([
      dao.calculateTax("yes", 2, 45000),
      dao.calculateTax("no", 2, 45000),
      dao.calculateTax("no", 1, 30000)
    ]);
    log(response);
    // Liste der Simulationen
    log("-----------list-of-simulations");
    response = await dao.listSimulations();
    log(response);
    // Löschen einer Simulation
    log("-----------deleting simulation no. 1");
    response = await dao.deleteSimulation(1);
    log(response);
  } catch (error: any) {
    // wir protokollieren den Fehler
    console.log("error=", error.message);
  }
}

// Geschäftsklasse
class Business {

  // [any]: die genaue Struktur der vom Steuerserver gesendeten Daten (Sätze, Obergrenzen, ...)
  // ist hier nicht dokumentiert; wir bleiben bei diesem Typ daher bewusst flexibel
  taxAdminData: any;

  // Konstruktor
  constructor(taxAdmindata: any) {
    // this.taxAdminData: Daten der Steuerverwaltung
    this.taxAdminData = taxAdmindata;
  }

  // Steuerberechnung
  // --------------------------------------------------------------------------
  calculateTax(married: string, children: number, salary: number): any {
    // married: ja, nein
    // children: Anzahl der Kinder
    // salary: Jahresgehalt
    // this.taxAdminData: Daten der Steuerverwaltung
    //
    // Steuerberechnung mit Kindern
    const result1 = this.calculateTax2(married, children, salary);
    const tax1 = result1["tax"];
    // Steuerberechnung ohne die Kinder
    let result2, tax2, halfShareCeiling;
    if (children !== 0) {
      result2 = this.calculateTax2(married, 0, salary);
      tax2 = result2["tax"];
      // Anwendung der Obergrenze des Familienquotienten
      halfShareCeiling = this.taxAdminData.halfShareQfCeiling;
      if (children < 3) {
        // HALF_SHARE_QF_CEILING Euro für die ersten 2 Kinder
        tax2 = tax2 - children * halfShareCeiling;
      } else {
        // HALF_SHARE_QF_CEILING Euro für die ersten 2 Kinder, doppelt für die folgenden
        tax2 = tax2 - 2 * halfShareCeiling - (children - 2) * 2 * halfShareCeiling;
      }
    } else {
      // keine Neuberechnung der Steuer
      tax2 = tax1;
      result2 = result1;
    }
    // wir nehmen die höhere Steuer aus [tax1, tax2]
    let tax, rate, surcharge;
    if (tax1 > tax2) {
      tax = tax1;
      rate = result1["rate"];
      surcharge = result1["surcharge"];
    } else {
      surcharge = tax2 - tax1 + result2["surcharge"];
      tax = tax2;
      rate = result2["rate"];
    }
    // Berechnung eines möglichen Nachlasses
    const discount = this.getDiscount(married, tax);
    tax -= discount;
    // Berechnung einer möglichen Steuerermäßigung
    const reduction = this.getReduction(married, salary, children, tax);
    tax -= reduction;
    // Ergebnis
    return {
      "tax": Math.floor(tax), "surcharge": surcharge, "discount": discount, "reduction": reduction,
      "rate": rate
    };
  }

  // --------------------------------------------------------------------------
  calculateTax2(married: string, children: number, salary: number): any {
    // married: ja, nein
    // children: Anzahl der Kinder
    // salary: Jahresgehalt
    // this->taxAdminData: Daten der Steuerverwaltung
    // 
    // Anzahl der Anteile
    married = married.toLowerCase();
    let shares;
    if (married === "yes") {
      shares = children / 2 + 2;
    } else {
      shares = children / 2 + 1;
    }
    // 1 Anteil pro Kind ab dem 3.
    if (children >= 3) {
      // ein weiterer halber Anteil für jedes Kind ab dem 3.
      shares += 0.5 * (children - 2);
    }
    // zu versteuerndes Einkommen
    const taxableIncome = this.getTaxableIncome(salary);
    // Zuschlag
    let surcharge = Math.floor(taxableIncome - 0.9 * salary);
    // wegen Rundungsproblemen
    if (surcharge < 0) {
      surcharge = 0;
    }
    // Familienquotient
    const quotient = taxableIncome / shares;
    // Steuerberechnung
    const limits = this.taxAdminData.limits;
    const coeffR = this.taxAdminData.coeffR;
    const coeffN = this.taxAdminData.coeffN;
    // wird ans Ende des Arrays limits gesetzt, um die folgende Schleife zu stoppen
    limits[limits.length - 1] = quotient;
    // Suche nach dem Steuersatz
    let i = 0;
    while (quotient > limits[i]) {
      i++;
    }
    // da wir quotient ans Ende des Arrays limits gesetzt haben, kann die vorherige Schleife
    // nicht über das Array limits hinausgehen
    // jetzt können wir die Steuer berechnen
    const tax = Math.floor(taxableIncome * coeffR[i] - shares * coeffN[i]);
    // Ergebnis
    return { "tax": tax, "surcharge": surcharge, "rate": coeffR[i] };
  }

  // taxableIncome=annualSalary-allowance
  // der Freibetrag hat ein Minimum und ein Maximum
  getTaxableIncome(salary: number): number {
    // Freibetrag von 10 % des Gehalts
    let allowance = 0.1 * salary;
    // dieser Freibetrag darf taxAdminData.getTenPercentAllowanceMax() nicht überschreiten
    if (allowance > this.taxAdminData.tenPercentAllowanceMax) {
      allowance = this.taxAdminData.tenPercentAllowanceMax;
    }
    // der Freibetrag darf taxAdminData.getTenPercentAllowanceMin() nicht unterschreiten
    if (allowance < this.taxAdminData.tenPercentAllowanceMin) {
      allowance = this.taxAdminData.tenPercentAllowanceMin;
    }
    // zu versteuerndes Einkommen
    const taxableIncome = salary - allowance;
    // Ergebnis
    return Math.floor(taxableIncome);
  }

  // berechnet einen möglichen Nachlass
  getDiscount(married: string, taxes: number): number {
    // zunächst ein Nachlass von null
    let discount = 0;
    // maximaler Steuerbetrag, um den Nachlass zu erhalten
    let taxCeilingForDiscount = married === "yes" ?
      this.taxAdminData.coupleTaxCeilingForDiscount :
      this.taxAdminData.singleTaxCeilingForDiscount;
    let discountCeiling;
    if (taxes < taxCeilingForDiscount) {
      // maximaler Betrag des Nachlasses
      discountCeiling = married === "yes" ?
        this.taxAdminData.coupleDiscountCeiling :
        this.taxAdminData.singleDiscountCeiling;
      // theoretischer Nachlass
      discount = discountCeiling - 0.75 * taxes;
      // der Nachlass darf den Betrag der Steuer nicht überschreiten
      if (discount > taxes) {
        discount = taxes;
      }
      // kein Nachlass <0
      if (discount < 0) {
        discount = 0;
      }
    }
    // Ergebnis
    return Math.ceil(discount);
  }

  // berechnet eine mögliche Ermäßigung
  getReduction(married: string, salary: number, children: number, taxes: number): number {
    // die Einkommensgrenze, um Anspruch auf die Ermäßigung von 20 % zu haben
    let incomeCeilingForReduction = married === "yes" ?
      this.taxAdminData.coupleIncomeCeilingForReduction :
      this.taxAdminData.singleIncomeCeilingForReduction;
    incomeCeilingForReduction += children * this.taxAdminData.halfShareReductionValue;
    if (children > 2) {
      incomeCeilingForReduction += (children - 2) * this.taxAdminData.halfShareReductionValue;
    }
    // zu versteuerndes Einkommen
    const taxableIncome = this.getTaxableIncome(salary);
    // Ermäßigung
    let reduction = 0;
    if (taxableIncome < incomeCeilingForReduction) {
      // Ermäßigung von 20 %
      reduction = 0.2 * taxes;
    }
    // Ergebnis
    return Math.ceil(reduction);
  }
}

      url: 'main.php',
      method: "GET",
      // URL-Parameter

        action: 'get-admindata'

    // Ergebnis

  async getRemoteData(options: AxiosRequestConfig): Promise<any> {
    // für das Session-Cookie
    if (!options.headers) {
      options.headers = {} as any;
    }
    (options.headers as any).Cookie = this.sessionCookie;
    // Ausführung der HTTP-Anfrage
    let response: AxiosResponse;
    try {
      // asynchrone Anfrage
      response = await this.axios.request(options);
    } catch (error: any) {
      // der Parameter [error] ist eine Exception-Instanz - sie kann verschiedene Formen annehmen
      if (error.response) {
        // die Antwort des Servers steht in [error.response]
        response = error.response;
      } else {
        // wir werfen den Fehler erneut
        throw error;
      }
    }
    // response ist die gesamte HTTP-Antwort des Servers (HTTP-Header + die eigentliche Antwort)
    // wir rufen das Session-Cookie ab, falls vorhanden
    const setCookie = response.headers['set-cookie'];
    if (setCookie) {
      // setCookie ist ein Array
      // wir suchen das Session-Cookie in diesem Array
      let found = false;
      let i = 0;
      while (!found && i < setCookie.length) {
        // wir suchen das Session-Cookie
        const results = RegExp('^(' + this.sessionCookieName + '.+?);').exec(setCookie[i]);
        if (results) {
          // wir speichern das Session-Cookie
          // eslint-disable-next-line require-atomic-updates
          this.sessionCookie = results[1];
          // wir haben es gefunden
          found = true;
        } else {
          // nächstes Element
          i++;
        }
      }
    }
    // die Antwort des Servers steht in [response.data]
    return response.data;
  }
}

    In diesem Abschnitt führen wir [HTTP client 2] als Anwendung in einem Browser aus, gemäß der folgenden Architektur:

    Während [node.js] TypeScript über tsx nativ ausführt, gilt das für Browser nicht: Es ist ein Werkzeug erforderlich, das TypeScript in JavaScript kompiliert und das Ganze anschließend zusammenfügt zu einer einzigen Datei, die von einer HTML-Seite verwendet werden kann. Das ist die Aufgabe von [webpack].

    Hinweis: Der oben beschriebene Vorgang [1] ist eine HTTP-Anfrage, die vom JavaScript-Code einer HTML-Seite erzeugt wird. Wir werden gleich sehen, dass diese HTML-Seite über die URL [http://localhost:8080] (oder 8081) bezogen wird, die an einen internen, von [webpack] gestarteten Server gerichtet ist. Der Steuerberechnungsserver hingegen ist eine NestJS-Anwendung, die unter der URL [http://localhost:3000] erreichbar ist (siehe Installation eines NestJS-Servers). Wir haben hier also zwei Webanwendungen, die miteinander kommunizieren und beide über [http://localhost] angesprochen werden, jedoch nicht über denselben Port. Man sagt, sie befinden sich in unterschiedlichen Domänen. Diese Art von Anfragen zwischen zwei Webanwendungen in unterschiedlichen Domänen wird kontrolliert. Standardmäßig lehnt der NestJS-Server Anfragen eines Web-Clients ab, wenn der Client nicht denselben Ursprung wie der Server hat, d. h. [http://localhost:3000]. Es ist jedoch möglich, den NestJS-Server so zu konfigurieren, dass er diese sogenannten „Cross-Domain“-Anfragen akzeptiert. Das ist hier geschehen. Deshalb funktioniert der unten gezeigte JavaScript-Client. Die Konfiguration des NestJS-Servers können Sie im Kapitel „Fallstudie: Der Steuerberechnungsserver in NestJS“ einsehen; dort wird gezeigt, wie er so konfiguriert wird, dass er Cross-Domain-Anfragen akzeptiert.

// Importe
import { AxiosInstance, AxiosRequestConfig, AxiosResponse } from 'axios';
import qs from 'qs'

class Dao3 {

  private axios: AxiosInstance;

  // constructor
  constructor(axios: AxiosInstance) {
    this.axios = axios;
  }

  // Sitzung initialisieren
  async initSession(): Promise<any> {
    // Optionen der HTTP-Anfrage [get /main.php?action=init-session&type=json]
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "GET",
      // URL-Parameter
      params: {
        action: 'init-session',
        type: 'json'
      }
    };
    // Ausführung der HTTP-Anfrage
    return await this.getRemoteData(options);
  }

  async authenticateUser(user: string, password: string): Promise<any> {
    // Optionen der HTTP-Anfrage [post /main.php?action=authenticate-user]
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "POST",
      headers: {
        'Content-type': 'application/x-www-form-urlencoded',
      },
      // Body des POST
      data: qs.stringify({
        user: user,
        password: password
      }),
      // URL-Parameter
      params: {
        action: 'authenticate-user'
      }
    };
    // Ausführung der HTTP-Anfrage
    return await this.getRemoteData(options);
  }

  async getAdminData(): Promise<any> {
    // Optionen der HTTP-Anfrage  [get /main.php?action=get-admindata]
    const options: AxiosRequestConfig = {
      url: 'main.php',
      method: "GET",
      // URL-Parameter
      params: {
        action: 'get-admindata'
      }
    };
    // Ausführung der HTTP-Anfrage
    const data = await this.getRemoteData(options);
    // result
    return data;
  }

  async getRemoteData(options: AxiosRequestConfig): Promise<any> {
    // Ausführung der HTTP-Anfrage
    let response: AxiosResponse;
    try {
      // asynchrone Anfrage
      response = await this.axios.request(options);
    } catch (error: any) {
      // der Parameter [error] ist eine Exception-Instanz - sie kann verschiedene Formen annehmen
      if (error.response) {
        // die Antwort des Servers steht in [error.response]
        response = error.response;
      } else {
        // wir werfen den Fehler erneut
        throw error;
      }
    }
    // response is the whole HTTP response of the server (HTTP headers + the response itself)
    // the server's response is in [response.data]
    return response.data;
  }
}

    6.5.1. Lösungsarchitektur

    Dieser HTML-Client arbeitet exakt wie [main3.ts], und das ist eine zentrale Entwurfsentscheidung: Er ruft den (PHP-)Server nur für drei Aktionen auf, und zwar immer dieselben: init-session, authenticate-user und get-admindata. Der (PHP-)Code des Servers zeigt jedoch, dass dieser auch selbst die Steuer berechnen und die Simulationen in seiner Session speichern kann (das sind die Aktionen calculate-tax, list-simulations und delete-simulation, die sich im Ordner [Controllers] finden), doch dieser Client nutzt sie nicht.

    Der Grund ist einfach: Sobald die Steuerdaten (Sätze, Schwellenwerte usw.) einmalig mit get-admindata abgerufen wurden, ist dank der Klasse [Business] bereits alles, was für die Steuerberechnung nötig ist, im Browser vorhanden. Diese Berechnung bei jeder Simulation vom Server anzufordern, würde einen vermeidbaren Netzwerk-Roundtrip erfordern; ebenso brächte es keinen Mehrwert, die Liste erneut vom Server anzufordern (list-simulations), da jede berechnete Simulation sofort im Browser gespeichert wird. Dieser Client minimiert die Serveranfragen daher so weit wie möglich – genau das Prinzip, das bereits in [main3.ts] umgesetzt ist:

    • die [dao]-Schicht ([Dao3.ts]) bleibt unverändert: Ihre drei bereits oben vorgestellten Methoden (initSession, authenticateUser, getAdminData) genügen; es wird keine hinzugefügt;
    • unmittelbar nach erfolgreicher Authentifizierung wird getAdminData ein für alle Mal aufgerufen, und mit dem Ergebnis wird ein Business-Objekt erzeugt (new Business(taxAdminData)), das während der gesamten Arbeitssitzung erhalten bleibt;
    • jede Steuerberechnung ruft anschließend direkt business.calculateTax(married, children, salary) auf – eine gewöhnliche JavaScript-Funktion, die im Browser ohne jeden Netzwerkaufruf ausgeführt wird;
    • die berechneten Simulationen werden in einem einfachen JavaScript-Array gespeichert, das der Controller im Speicher hält – ein push pro Berechnung, ein splice pro Löschung, genau wie es [main3.ts] mit den Anweisungen simulations.push(...) und simulations.splice(1, 1) tut.

    Ein kleiner Unterschied zu [main3.ts]: Die Methode business.calculateTax liefert nur die von ihr berechneten Werte zurück (tax, surcharge, discount, reduction, rate), nicht die vom Benutzer eingegebenen Parameter (married, children, salary). [main3.ts] braucht sie nicht, da es einfach console.log verwendet; unsere View [Liste der Simulationen] muss diese drei Parameter jedoch in der Tabelle anzeigen – daher hängt der Controller sie vor dem Speichern selbst an das Ergebnis an (das sehen wir im Code von [AppHtml.ts]).

    Schließlich ruft auch die View [Sitzungsende] den Server nicht mehr auf: Sie verwirft lediglich das [Business]-Objekt und die Liste der Simulationen und zeigt dann erneut den Authentifizierungsbildschirm an. Zur Erinnerung: Die Antworten der drei weiterhin verwendeten Serveraktionen haben immer dasselbe Format, das wir bereits in [main3.ts] gesehen haben: ein JSON-Objekt { action, status, response }, wobei response entweder das angeforderte Ergebnis (bei Erfolg) oder eine Fehlermeldung enthält – je nach zuständigem PHP-Controller ein String oder ein Array von Strings.

    6.5.1.4. Die [views]-Schicht: die Datei [Views.ts]

    Dies ist die längste Datei dieses Clients, aber auch die am leichtesten verständliche, sobald man ihr Prinzip erfasst hat: Die Klasse [Views] führt keinerlei Netzwerkaufrufe aus – sie weiß nicht einmal, dass der PHP-Server existiert. Ihr einziger Zweck ist es, mit den DOM-Funktionen (document.createElement, appendChild, classList ...) die HTML-Elemente der drei Bildschirme der Anwendung innerhalb des <div id="app"> -Tags des Templates zu erzeugen.

    Wenn der Benutzer auf eine Schaltfläche oder einen Link klickt („Submit“, „Delete“, „End of session“ usw.), weiß die Klasse [Views] mit diesem Klick ebenfalls nichts anzufangen: Sie ruft lediglich eine vom Controller bereitgestellte Callback-Funktion auf – der Controller, [AppHtml.ts], der kurz darauf beschrieben wird, entscheidet dann, was zu tun ist (den Server aufrufen, die View wechseln usw.). Diese Entkopplung ermöglicht es [Views.ts] , unabhängig von der [dao]-Schicht zu bleiben.

    6.5.1.5. Der Controller: die Datei [AppHtml.ts]

    Dies ist der Orchestrator der Anwendung: Er übernimmt die Rolle der Funktion main() aus **[main3.ts], jedoch in Form einer Klasse mit einer Methode pro View. Er koordiniert nun zwei Schichten, [dao] (der Server) und [business] (die Berechnung), bevor er die [views]**-Schicht anweist, entweder ein Ergebnis oder eine Fehlermeldung anzuzeigen:

    
// Importe
import axios from 'axios';
import "core-js/stable";
import "regenerator-runtime/runtime";

// Importe
import Dao from './Dao3';
import Business from './Business';
import Views, { Simulation } from './Views';

// -----------------------------------------------------------------------
// Controller der HTML-Anwendung: orchestriert die Aufrufe des Servers
// (Schicht [dao]), die Berechnungen (Schicht [business]) und die Anzeige der Views
// (Schicht [views]).
//
// Wie [main3.ts] ruft dieser Client den PHP-Server nur für drei
// Aktionen auf: [init-session], [authenticate-user] und [get-admindata].
// Sobald die Steuerdaten (Sätze, Obergrenzen ...) mit
// [get-admindata] abgerufen wurden, berechnet die Schicht [business] (Klasse [Business], dieselbe
// Datei [Business.ts] wie bei [main3.ts]) die Steuer im
// Browser, ohne einen weiteren Netzwerkaufruf. Auch die Liste der Simulationen wird
// nur im Browser gehalten (Array [this.simulations]),
// genau wie [main3.ts] es mit seinem lokalen Array [simulations] tut
// (siehe `simulations.push(...)` und `simulations.splice(...)`).
// -----------------------------------------------------------------------
class AppHtml {

  private dao: Dao;
  private views: Views;
  // [business]-Schicht: existiert erst, wenn die Steuerdaten abgerufen wurden
  // (nach erfolgreicher Authentifizierung) - siehe [initBusiness]
  private business?: Business;
  // Liste der bereits berechneten Simulationen, im Browser gehalten
  private simulations: Simulation[] = [];

  constructor() {
    // axios-Konfiguration - identisch mit [main3.ts]
    axios.defaults.timeout = 5000;
    axios.defaults.baseURL = 'http://localhost/';
    axios.defaults.withCredentials = true;
    // [dao]- und [views]-Schichten
    this.dao = new Dao(axios);
    this.views = new Views(document.getElementById('app')!);
  }

  // Einstiegspunkt der Anwendung
  async start(): Promise<void> {
    try {
      // Sitzung initialisieren - zwingend vor jeder anderen Aktion (siehe [main.php])
      const response = await this.dao.initSession();
      if (response.status !== 700) {
        this.views.authenticationView(this.formatError(response.response), {
          onSubmit: (user, password) => this.authenticate(user, password)
        });
        return;
      }
    } catch (error: any) {
      this.views.authenticationView(
        "Unable to contact the server (" + error.message + ")",
        { onSubmit: (user, password) => this.authenticate(user, password) }
      );
      return;
    }
    // Sitzung initialisiert: Wir zeigen die Authentifizierung an
    this.showAuthentication();
  }

  // --------------------------------------------------------------------
  // View [authentication]
  // --------------------------------------------------------------------
  private showAuthentication(error?: string): void {
    this.views.authenticationView(error, {
      onSubmit: (user, password) => this.authenticate(user, password)
    });
  }

  private async authenticate(user: string, password: string): Promise<void> {
    try {
      const response = await this.dao.authenticateUser(user, password);
      if (response.status === 200) {
        // erfolgreiche Authentifizierung: Wie in [main3.ts] rufen wir nun
        // ein für alle Mal die Steuerdaten ab,
        // die von der [business]-Schicht benötigt werden
        await this.initBusiness();
      } else {
        // Fehlschlag (falsches Login/Passwort oder fehlende Parameter)
        this.showAuthentication(this.formatError(response.response));
      }
    } catch (error: any) {
      this.showAuthentication("Unable to contact the server (" + error.message + ")");
    }
  }

  // ruft die Daten der Steuerverwaltung ab und instanziiert die
  // [business]-Schicht mit diesen Daten - ein einziger Serveraufruf für die gesamte Sitzung,
  // genau wie [main3.ts]
  private async initBusiness(): Promise<void> {
    try {
      const response = await this.dao.getAdminData();
      if (response.status === 1000) {
        this.business = new Business(response.response);
        // neue Arbeitssitzung: Wir beginnen wieder mit einer leeren Liste
        this.simulations = [];
        this.showCalculation();
      } else {
        this.showAuthentication(this.formatError(response.response));
      }
    } catch (error: any) {
      this.showAuthentication("Unable to contact the server (" + error.message + ")");
    }
  }

  // --------------------------------------------------------------------
  // View [tax calculation]
  // --------------------------------------------------------------------
  private showCalculation(result?: Simulation, error?: string): void {
    this.views.calculationView(result, error, {
      onSubmit: (married, children, salary) => this.calculate(married, children, salary),
      onList: () => this.showList(),
      onEndSession: () => this.endSession()
    });
  }

  // Steuerberechnung vollständig im Browser ([business]-Schicht):
  // kein Serveraufruf, anders als in der ersten Version dieses Clients
  private calculate(married: string, children: number, salary: number): void {
    // this.business wurde zwangsläufig initialisiert: Wir gelangen erst nach [initBusiness]
    // zur View [calculation]
    const result = this.business!.calculateTax(married, children, salary);
    // Das von [business.calculateTax] zurückgegebene Ergebnis enthält nur die
    // berechneten Werte (tax, surcharge, discount, reduction, rate): Wir ergänzen es um die
    // vom Benutzer eingegebenen Parameter, damit sie
    // erneut in der Liste der Simulationen angezeigt werden können
    const simulation: Simulation = { married: married, children: children, salary: salary, ...result };
    // Speicherung im Browser (wie `simulations.push(...)` in [main3.ts])
    this.simulations.push(simulation);
    this.showCalculation(simulation);
  }

  // --------------------------------------------------------------------
  // View [list of simulations]
  // --------------------------------------------------------------------
  // keine Serveraufrufe mehr hier: Die Liste befindet sich bereits vollständig in
  // [this.simulations]
  private showList(error?: string): void {
    this.views.listView(this.simulations, error, {
      onDelete: (number) => this.deleteSimulation(number),
      onCalculation: () => this.showCalculation(),
      onEndSession: () => this.endSession()
    });
  }

  private deleteSimulation(number: number): void {
    // lokales Löschen aus dem Array [this.simulations], wie
    // `simulations.splice(1, 1)` in [main3.ts] - weiterhin kein Serveraufruf
    this.simulations.splice(number, 1);
    this.showList();
  }

  // --------------------------------------------------------------------
  // Ende der Sitzung
  // --------------------------------------------------------------------
  private endSession(): void {
    // Ende der Sitzung nur auf der Client-Seite: Wir vergessen die [business]-Schicht
    // (und damit die Steuerdaten) und die bereits berechneten Simulationen und
    // kehren dann zum Authentifizierungsbildschirm zurück. Dieser Client muss dafür
    // den Server nicht aufrufen.
    this.business = undefined;
    this.simulations = [];
    this.showAuthentication();
  }

  // --------------------------------------------------------------------
  // formatiert das Feld [response] einer Fehlerantwort des Servers,
  // das entweder ein String oder ein Array von Strings ist (siehe PHP-Controller)
  // --------------------------------------------------------------------
  private formatError(response: unknown): string {
    if (Array.isArray(response)) {
      return response.join(' ; ');
    }
    return String(response);
  }
}

window.addEventListener('DOMContentLoaded', () => {
  new AppHtml().start();
});

    • Zeilen 28 und 29: DAO und views, bereits in der ersten Version dieses Clients vorhanden;
    • Zeilen 30–32: neues Feld business vom Typ Business – das Fragezeichen (?) macht es zu einem optionalen Feld: Solange der Benutzer nicht authentifiziert ist, hat diese Eigenschaft noch keinen Wert (sie ist undefined);

    • Zeile 50: Das Allererste, was die Anwendung tut, ist der Aufruf von init-session, der vor jeder anderen Aktion erforderlich ist (siehe die PHP-Datei [main.php], die jede Aktion blockiert, bis eine Session initialisiert wurde);

    • showAuthentication (Zeilen 71–75): unverändert; die Methode weist lediglich die [views]-Schicht an, diese View anzuzeigen, wobei die Methode authenticate über onSubmit als Callback dient;

    • initBusiness (Zeilen 97–111), eine neue Methode: Sie ruft dao.getAdminData() auf – dies ist die einzige Stelle in der gesamten Datei, an der diese Aktion beim Server angefordert wird, und zwar ein für alle Mal; lautet der Status 1000, erzeugt sie das Business-Objekt mit den empfangenen Daten (Zeile 101, new Business(response.response)), setzt das Simulations-Array zurück (Zeile 103 – eine neue Authentifizierung entspricht einer neuen Arbeitssitzung) und zeigt schließlich die Berechnungs-View an (Zeile 104).

    Die View für die Steuerberechnung (Zeilen 116–138):

    • showCalculation (Zeilen 116–122): unverändert – sie kann mit oder ohne Ergebnis aufgerufen werden: ohne Ergebnis für ein leeres Formular (erster Besuch oder Rückkehr aus der Liste), mit Ergebnis unmittelbar nach einer erfolgreichen Berechnung;
    • calculate (Zeilen 126–138): Dies ist die Methode, die sich am stärksten geändert hat. Sie ist nicht mehr asynchron und enthält keinen try/catch-Block für Netzwerkfehler mehr, da sie keine Serveraufrufe mehr ausführt; in Zeile 129 berechnet this.business!.calculateTax(married, children, salary) direkt das Ergebnis – das Ausrufezeichen (!) teilt TypeScript mit, dass wir sicher sind, dass business bereits initialisiert wurde (diese View ist nur nach initBusiness erreichbar); in den Zeilen 130 bis 134 werden die eingegebenen Parameter, die dieses Ergebnis nicht enthält, mit dem Spread-Operator ...result angehängt, um das vollständige simulation-Objekt zu bilden; in Zeile 136 speichert this.simulations.push(simulation) diese neue Simulation im Array des Browsers; in Zeile 137 wird die View schließlich mit diesem Ergebnis neu aufgebaut.

    • showList (Zeilen 145–151): führt keine Serveraufrufe mehr aus – sie übergibt direkt this.simulations, das bereits im Speicher vorhandene Array, an die [views]-Schicht;
    • deleteSimulation (Zeilen 153–158): In Zeile 156 entfernt this.simulations.splice(number, 1) die Simulation direkt aus dem Array des Browsers – das exakte Äquivalent der Anweisung simulations.splice(1, 1) in [main3.ts] –, danach wird in Zeile 157 die (aktualisierte) Liste erneut angezeigt.

    Sitzungsende und Einstiegspunkt (Zeilen 163–189):

    • endSession (Zeilen 163–171): Die Methode ruft nicht mehr dao.endSession() auf (diese Methode wurde aus [Dao3.ts] entfernt) – sie verwirft einfach die [business]-Schicht (Zeile 168, this.business = undefined) und die Simulationen (Zeile 169, this.simulations = []) und zeigt dann den Authentifizierungsbildschirm erneut an (Zeile 170); wie der Rest dieses Clients benötigt auch das Beenden der Sitzung daher keine Netzwerkanfragen mehr;
    • formatError (Zeilen 177–182): unverändert – das Feld response eines Serverfehlers (PHP) ist manchmal ein String und manchmal ein Array von Strings (je nach Controller); diese Methode reduziert die Meldung immer auf einen einzigen String, der zur Anzeige bereit ist;
    • Zeilen 186–188: Der Einstiegspunkt des Skripts außerhalb der Klasse bleibt unverändert – DOMContentLoaded stellt sicher, dass die HTML-Seite vollständig geladen ist, bevor der Controller erzeugt und start() aufgerufen wird; dies ist für diesen GUI-Client das Äquivalent zum Aufruf main(); am Ende von [main3.ts].
